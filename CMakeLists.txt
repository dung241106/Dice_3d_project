cmake_minimum_required(VERSION 3.16)
project(Dice3DEducation VERSION 2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenGL REQUIRED)
find_package(GLUT REQUIRED)
add_executable(Dice3D_Edu src/main.cpp src/ui.cpp src/view.cpp src/storage.cpp src/dice.cpp src/lab.cpp src/puzzle.cpp src/throw_sim.cpp src/sound.cpp src/tests.cpp src/throw_tests.cpp)
target_include_directories(Dice3D_Edu PRIVATE include)
target_link_libraries(Dice3D_Edu PRIVATE OpenGL::GL OpenGL::GLU GLUT::GLUT)
if(WIN32)
    target_link_libraries(Dice3D_Edu PRIVATE gdi32 winmm)
endif()
if(MSVC)
    target_compile_options(Dice3D_Edu PRIVATE /W4 /utf-8)
else()
    target_compile_options(Dice3D_Edu PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME math_physics_puzzle COMMAND Dice3D_Edu --test)
